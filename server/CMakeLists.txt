cmake_minimum_required(VERSION 3.16)
project(smart_im LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost 1.74 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../data/cpp-headers" "${DEPS_ROOT}/include" REQUIRED)
endif()
find_path(MYSQL_INCLUDE_DIR NAMES mysql.h HINTS "${MYSQL_ROOT}/include" PATH_SUFFIXES mysql mariadb REQUIRED)
find_library(MYSQL_LIBRARY NAMES libmysql mysqlclient mariadb libmariadb HINTS "${MYSQL_ROOT}/lib" REQUIRED)
add_executable(im_server main.cpp)
target_include_directories(im_server PRIVATE ${Boost_INCLUDE_DIRS} ${MYSQL_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
target_compile_definitions(im_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_ALL_NO_LIB)
if(MSVC)
  target_compile_options(im_server PRIVATE /W4 /utf-8 /bigobj /permissive-)
  target_compile_definitions(im_server PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(im_server PRIVATE ws2_32 mswsock)
else()
  target_compile_options(im_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_link_libraries(im_server PRIVATE OpenSSL::Crypto Threads::Threads ${MYSQL_LIBRARY})
if(nlohmann_json_FOUND)
  target_link_libraries(im_server PRIVATE nlohmann_json::nlohmann_json)
endif()
